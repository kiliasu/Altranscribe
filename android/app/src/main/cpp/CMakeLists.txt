cmake_minimum_required(VERSION 3.22.1)
project(altranscribe_audio LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(REPOSITORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
add_subdirectory("${REPOSITORY}/native/third_party" audio_dependencies)
add_library(altranscribe_audio SHARED audio_jni.cpp "${REPOSITORY}/native/audio/audio_processing.cpp")
target_include_directories(altranscribe_audio PRIVATE "${REPOSITORY}/native/audio")
target_link_libraries(altranscribe_audio PRIVATE alt_rnnoise alt_resampler)
target_compile_options(altranscribe_audio PRIVATE -O2)
target_compile_options(alt_rnnoise PRIVATE -O2)
target_compile_options(alt_resampler PRIVATE -O2)

set(FFMPEG "${REPOSITORY}/.tools/ffmpeg-android/${ANDROID_ABI}")
if(NOT EXISTS "${FFMPEG}/lib/libavformat.so")
  message(FATAL_ERROR "Run scripts/setup-android-ffmpeg.ps1 before building Android.")
endif()
foreach(component avformat avcodec swresample avutil)
  add_library(${component} SHARED IMPORTED)
  set_target_properties(${component} PROPERTIES IMPORTED_LOCATION "${FFMPEG}/lib/lib${component}.so")
endforeach()
add_library(altranscribe_media SHARED media_jni.cpp)
target_include_directories(altranscribe_media PRIVATE "${FFMPEG}/include")
target_link_libraries(altranscribe_media PRIVATE avformat avcodec swresample avutil)
